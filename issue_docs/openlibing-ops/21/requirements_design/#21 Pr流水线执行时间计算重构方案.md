"""
Pipeline效率时长计算脚本
=======================
基于dwi_rd_efc_pipeline_run_job表，计算每次pipeline_run的效率相关时长。

计算逻辑：
  1. 分别计算每个stage的时间，多个stage时间相加为最终的效率时长
  2. 每个stage的时间由内部的job任务耗时计算，通过stage_id进行分组
  3. stage中job可能存在依赖关系（dependsOn引用define_id），存在依赖关系的job耗时相加
  4. stage中存在依赖关系的任务链和不存在依赖关系的独立任务，取最大时长作为stage耗时

依赖关系说明：
  dependsOn字段存储单个define_id，形成树/链式依赖关系
  通过构建DAG图，从每个job向上追溯依赖链，计算关键路径时长
  stage时长 = 关键路径时长 = MAX(所有链的累计时长)

数据模型示例：
  Job A (dependsOn=null)  Job D (dependsOn=null)
      |                        |
  Job B (dependsOn=A)         |
      |                        |
  Job C (dependsOn=B)         |

  链1: A→B→C  chain_duration = dur(A)+dur(B)+dur(C)
  链2: D      chain_duration = dur(D)
  stage时长 = MAX(chain1_duration, chain2_duration)
  pipeline效率时长 = SUM(各stage时长)

内存优化：
  按pipeline_run_id分批查询，每批只处理一批pipeline_run的数据
  处理完立即写入结果并释放内存，避免全量加载

多线程：
  多线程并发处理不同批次的pipeline_run（查询+计算）
  写入结果时使用线程锁保证顺序

使用方式：
  python pipeline_efficiency_duration.py --host <host> --port 9030 --user root --password <pwd> --database <db>
  python pipeline_efficiency_duration.py --config config.json

def calc_job_duration_sec(start_time, end_time):
    if start_time is None or end_time is None:
        return 0
    if isinstance(start_time, str):
        start_time = datetime.fromisoformat(start_time.replace('Z', '+00:00'))
    if isinstance(end_time, str):
        end_time = datetime.fromisoformat(end_time.replace('Z', '+00:00'))
    delta = (end_time - start_time).total_seconds()
    return max(delta, 0)


def build_dependency_graph(jobs):
    define_map = {}
    for job in jobs:
        key = (job['project_id'], job['pipeline_id'], job['pipeline_run_id'], job['stage_id'], job['job_identifier'])
        define_map[key] = job

    memo = {}

    def critical_path(job_key, visited=None):
        if job_key in memo:
            return memo[job_key]
        if visited is None:
            visited = set()
        if job_key in visited:
            return 0
        visited.add(job_key)

        job = define_map.get(job_key)
        if job is None:
            return 0

        dep_str = (job.get('dependsOn') or '').strip()
        job_dur = job['job_duration_sec']

        if not dep_str:
            memo[job_key] = job_dur
            return job_dur

        dep_key = (job_key[0], job_key[1], job_key[2], job_key[3], dep_str)
        dep_dur = critical_path(dep_key, visited)

        memo[job_key] = job_dur + dep_dur
        return memo[job_key]

    for key in define_map:
        critical_path(key)

    return memo


def compute_efficiency(rows):
    stage_jobs = defaultdict(list)
    for row in rows:
        row['job_duration_sec'] = calc_job_duration_sec(row.get('job_start_time'), row.get('job_end_time'))
        stage_key = (row['project_id'], row['pipeline_id'], row['pipeline_run_id'], row['stage_id'])
        stage_jobs[stage_key].append(row)

    result = defaultdict(lambda: {'stages': {}, 'stage_details': {}})

    for stage_key, jobs in stage_jobs.items():
        project_id, pipeline_id, pipeline_run_id, stage_id = stage_key
        pr_key = (project_id, pipeline_id, pipeline_run_id)

        cp_map = build_dependency_graph(jobs)

        max_path = 0
        chain_details = []
        for job in jobs:
            job_key = (project_id, pipeline_id, pipeline_run_id, stage_id, job['job_identifier'])
            cp = cp_map.get(job_key, 0)
            chain_details.append({
                'job_name': job.get('job_name', ''),
                'job_identifier': job.get('job_identifier', ''),
                'dependsOn': (job.get('dependsOn') or '').strip(),
                'job_duration_sec': job['job_duration_sec'],
                'critical_path_sec': cp,
            })
            if cp > max_path:
                max_path = cp

        result[pr_key]['stages'][stage_id] = max_path
        result[pr_key]['stage_details'][stage_id] = chain_details

    for pr_key in result:
        result[pr_key]['efficiency_duration_sec'] = sum(result[pr_key]['stages'].values())

    return result


def create_connection(host, port, user, password, database):
    if pymysql is None:
        print("Error: pymysql is required. Install with: pip install pymysql", file=sys.stderr)

    return pymysql.connect(

        database=database, charset='utf8mb4',
        cursorclass=pymysql.cursors.DictCursor

def fetch_pipeline_run_ids(host, port, user, password, database, where_clause=None):
    query = PIPELINE_RUN_IDS_QUERY
    if where_clause:
        query += f" AND ({where_clause})"
    conn = create_connection(host, port, user, password, database)
    try:
        with conn.cursor() as cursor:
            cursor.execute(query)
            rows = cursor.fetchall()
            return [r['pipeline_run_id'] for r in rows]
    finally:
        conn.close()


def fetch_pipeline_info_batch(host, port, user, password, database, run_ids):
    if not run_ids:
        return {}
    placeholders = ','.join(['%s'] * len(run_ids))
    query = PIPELINE_INFO_QUERY.format(placeholders)
    conn = create_connection(host, port, user, password, database)
    try:
        with conn.cursor() as cursor:
            cursor.execute(query, list(run_ids))
            rows = cursor.fetchall()
            info_map = {}
            for row in rows:
                key = (row['project_id'], row['pipeline_id'], row['pipeline_run_id'])
                info_map[key] = row
            return info_map
    finally:
        conn.close()


def process_batch(batch_ids, host, port, user, password, database, insert_table=None):
    """
    处理一批pipeline_run_id：查询job数据 → 计算效率时长 → 可选写入Doris
    每批独立连接、独立计算、处理完即释放内存
    """
    placeholders = ','.join(['%s'] * len(batch_ids))
    query = BATCH_JOB_QUERY.format(placeholders)

    conn = create_connection(host, port, user, password, database)
    try:
        with conn.cursor() as cursor:
            cursor.execute(query, list(batch_ids))
            rows = cursor.fetchall()
    finally:
        conn.close()

    if not rows:
        return {}

    results = compute_efficiency(rows)

    pipeline_info = fetch_pipeline_info_batch(host, port, user, password, database, batch_ids)

    if insert_table:
        insert_results_batch(host, port, user, password, database, insert_table, results, pipeline_info)

    return dict(results), dict(pipeline_info)


def insert_results_batch(host, port, user, password, database, insert_table, results, pipeline_info):
    conn = create_connection(host, port, user, password, database)
    try:
        with conn.cursor() as cursor:
            sql = INSERT_SQL_TEMPLATE.format(insert_table)
            count = 0
            for pr_key, eff in results.items():
                project_id, pipeline_id, pipeline_run_id = pr_key
                eff_sec = eff['efficiency_duration_sec']
                info = (pipeline_info or {}).get(pr_key, {})
                start = info.get('pipeline_start_time')
                end = info.get('pipeline_end_time')
                wall_sec = calc_job_duration_sec(start, end) if start and end else None
                rate = 0
                if wall_sec and wall_sec > 0 and eff_sec > 0:
                    rate = round(eff_sec * 100.0 / wall_sec, 2)
                cursor.execute(sql, (project_id, pipeline_id, pipeline_run_id, eff_sec, rate))
                count += 1
            conn.commit()
            safe_print(f"  Inserted {count} records into {insert_table}")
    finally:
        conn.close()


def format_duration(sec):
    if sec is None:
        return '-'
    s = int(sec)
    if s < 60:
        return f"{s}s"
    m, rs = divmod(s, 60)
    if m < 60:
        return f"{m}m {rs}s"
    h, rm = divmod(m, 60)
    return f"{h}h {rm}m"

    print(f"Total: {len(all_results)} pipeline runs")


def print_stage_details(results):
    print(f"\n{'='*100}")
    print("Stage-level Details (Critical Path Analysis)")
    print(f"{'-'*100}")

    for pr_key in sorted(results.keys()):
        project_id, pipeline_id, pipeline_run_id = pr_key
        eff = results[pr_key]
        pr_label = pipeline_run_id[:20] if pipeline_run_id else '-'
        print(f"\nPipeline Run: {pr_label} | Efficiency Duration: {format_duration(eff['efficiency_duration_sec'])}")

        for stage_id, stage_dur in eff['stages'].items():
            print(f"  Stage: {stage_id[:20]}  Duration: {format_duration(stage_dur)}")
            details = eff['stage_details'].get(stage_id, [])

def main():
    parser = argparse.ArgumentParser(description='Pipeline效率时长计算脚本（分批+多线程版）')
    parser.add_argument('--host', default='localhost', help='Doris host')
    parser.add_argument('--port', type=int, default=9030, help='Doris port')
    parser.add_argument('--user', default='root', help='Doris user')
    parser.add_argument('--password', default='', help='Doris password')
    parser.add_argument('--database', default='', help='Doris database')
    parser.add_argument('--config', default=None, help='JSON config file path')
    parser.add_argument('--where', default=None, help='Additional WHERE clause for job query filter')
    parser.add_argument('--batch-size', type=int, default=50, help='Number of pipeline_run_ids per batch (default: 50)')
    parser.add_argument('--threads', type=int, default=4, help='Number of concurrent threads (default: 4)')
    parser.add_argument('--dry-run', action='store_true', help='Run with built-in test data')
    parser.add_argument('--detail', action='store_true', help='Show stage-level critical path details')
    parser.add_argument('--output', default=None, help='Output JSON file path')
    parser.add_argument('--insert', action='store_true', help='Insert results back to Doris')
    parser.add_argument('--insert-table', default='dwi_rd_efc_pipeline_run_efficiency', help='Target table for insert')
    args = parser.parse_args()

    if args.config:
        with open(args.config, 'r', encoding='utf-8') as f:
            cfg = json.load(f)
        for k in ('host', 'port', 'user', 'password', 'database', 'where', 'batch_size', 'threads'):
            if k in cfg:
                if k in ('port', 'batch_size', 'threads'):
                    setattr(args, k, int(cfg[k]))
                elif k == 'where':
                    args.where = cfg[k]
                else:
                    setattr(args, k, cfg[k])

    if args.dry_run:
        print("Running in DRY-RUN mode with built-in test data...\n")
        rows = generate_test_data()
        results = compute_efficiency(rows)
        print_result_table(dict(results), {})
        if args.detail:
            print_stage_details(dict(results))
        if args.output:
            write_output_json(args.output, results, {})
        return results

    if not args.database:
        print("Error: --database is required (or use --dry-run)", file=sys.stderr)
        sys.exit(1)

    print(f"Config: batch_size={args.batch_size}, threads={args.threads}")
    print(f"Fetching pipeline_run_ids from Doris...")

    all_run_ids = fetch_pipeline_run_ids(
        args.host, args.port, args.user, args.password, args.database, args.where
    )
    total = len(all_run_ids)
    print(f"Found {total} pipeline_run_ids")

    if total == 0:
        print("No data to process.")
        return {}

    batches = [all_run_ids[i:i + args.batch_size] for i in range(0, total, args.batch_size)]
    num_batches = len(batches)
    print(f"Split into {num_batches} batches (batch_size={args.batch_size})")

    all_results = {}
    all_pipeline_info = {}
    completed_count = 0
    start_time = time.time()

    with ThreadPoolExecutor(max_workers=args.threads) as executor:
        futures = {}
        for batch_idx, batch_ids in enumerate(batches):
            insert_table = args.insert_table if args.insert else None
            future = executor.submit(
                process_batch,
                batch_ids, args.host, args.port, args.user, args.password, args.database,
                insert_table
            )
            futures[future] = batch_idx

        for future in as_completed(futures):
            batch_idx = futures[future]
            try:
                batch_results, batch_info = future.result()
                with result_lock:
                    all_results.update(batch_results)
                    all_pipeline_info.update(batch_info)
                completed_count += 1
                elapsed = time.time() - start_time
                speed = completed_count / elapsed if elapsed > 0 else 0
                eta = (num_batches - completed_count) / speed if speed > 0 else 0
                safe_print(f"  Batch {completed_count}/{num_batches} done "
                           f"({len(batch_results)} runs) | "
                           f"elapsed={elapsed:.1f}s eta={eta:.1f}s")
            except Exception as e:
                safe_print(f"  Batch {batch_idx} failed: {e}", file=sys.stderr)

    elapsed_total = time.time() - start_time
    print(f"\nAll {num_batches} batches completed in {elapsed_total:.1f}s")
    print(f"Total pipeline runs processed: {len(all_results)}")

    print_result_table(all_results, all_pipeline_info)

    if args.detail:
        print_stage_details(all_results)

    if args.output:
        write_output_json(args.output, all_results, all_pipeline_info)

    if args.insert:
        print(f"\nResults have been inserted into {args.insert_table} per batch.")

    return all_results


def write_output_json(path, results, pipeline_info):
    output_data = []
    for pr_key, eff in results.items():
        project_id, pipeline_id, pipeline_run_id = pr_key
        info = (pipeline_info or {}).get(pr_key, {})
        output_data.append({
            'project_id': project_id,
            'pipeline_id': pipeline_id,
            'pipeline_run_id': pipeline_run_id,
            'pipeline_name': info.get('pipeline_name', ''),
            'pipeline_run_number': info.get('pipeline_run_number'),
            'efficiency_duration_sec': eff['efficiency_duration_sec'],
            'stage_count': len(eff['stages']),
            'stages': {sid: dur for sid, dur in eff['stages'].items()},
        })
    with open(path, 'w', encoding='utf-8') as f:
        json.dump(output_data, f, indent=2, default=str, ensure_ascii=False)
    print(f"\nResults written to {path}")


if __name__ == '__main__':
    main()
